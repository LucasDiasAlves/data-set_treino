import matplotlib.pyplot as plt
from pyspark.sql.functions import desc, sum

# 1. SPARK FAZ O TRABALHO PESADO
# Agrupamos por exercício e somamos a carga, depois pegamos o Top 5
df_resumo_spark = df_cru.groupBy("nome_exercicio") \
    .agg(sum("carga").alias("carga_total")) \
    .orderBy(desc("carga_total")) \
    .limit(5)

# 2. A PONTE PARA O MUNDO LOCAL (O Pulo do Gato)
# Trazemos APENAS as 5 linhas de resultado para o Pandas
df_pandas = df_resumo_spark.toPandas()

# 3. MATPLOTLIB FAZ A MÁGICA VISUAL (Usando sintaxe normal de Pandas)
plt.figure(figsize=(10, 5))
plt.bar(df_pandas["nome_exercicio"], df_pandas["carga_total"], color='royalblue')

plt.title("Top 5 Exercícios com Maior Carga Total", fontsize=14)
plt.xlabel("Exercício", fontsize=12)
plt.ylabel("Carga Total (kg)", fontsize=12)
plt.xticks(rotation=45)

# Exibe o gráfico na tela do Jupyter
plt.show()
